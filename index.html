Пример кода

Функция разбивает строку на пары символов и дополняет нечётную пару символом подчёркивания.

function splitString(str) {
  const result = [];

  for (let index = 0; index < str.length; index += 2) {
    const pair = str.slice(index, index + 2);
    result.push(pair.padEnd(2, '_'));
  }

  return result;
}
